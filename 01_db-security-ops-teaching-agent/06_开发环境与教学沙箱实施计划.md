# 开发环境与教学沙箱 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在不触碰本机 MySQL57 的前提下，建立可由 VS Code 使用、能够自动初始化和重置、并通过权限隔离测试的 MySQL 教学沙箱与容器化开发环境。

**Architecture:** Docker Compose 编排 `workspace` 开发容器和 `mysql-sandbox` 数据库容器；VS Code 通过 Dev Container 连接 `workspace`。MySQL 8.4 LTS 仅绑定 `127.0.0.1:3307`，教学异常以合成数据、权限快照和不可执行日志呈现，环境管理统一由 `scripts/sandbox.ps1` 完成。

**Tech Stack:** Docker Desktop、Docker Compose v2、MySQL 8.4 LTS、Python 3.13、Node.js 24 LTS、pytest 9.1.1、PyYAML 6.0.3、PyMySQL 1.2.0、PowerShell、VS Code Dev Containers。

Create `infra/mysql/init/001_schema.sql` with the exact statements below:

```sql
CREATE DATABASE IF NOT EXISTS shuqi_sandbox DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;
USE shuqi_sandbox;
CREATE TABLE IF NOT EXISTS citizen_profile (id BIGINT PRIMARY KEY, business_code VARCHAR(32) NOT NULL UNIQUE, full_name VARCHAR(64) NOT NULL, id_token VARCHAR(32) NOT NULL, phone VARCHAR(32) NOT NULL, created_at DATETIME NOT NULL) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS service_request (id BIGINT PRIMARY KEY, citizen_id BIGINT NOT NULL, request_type VARCHAR(64) NOT NULL, status VARCHAR(32) NOT NULL, submitted_at DATETIME NOT NULL, CONSTRAINT fk_request_citizen FOREIGN KEY (citizen_id) REFERENCES citizen_profile(id)) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS account_inventory (account_name VARCHAR(64) PRIMARY KEY, source_scope VARCHAR(128) NOT NULL, role_name VARCHAR(64) NOT NULL, status VARCHAR(32) NOT NULL, last_active_at DATETIME NULL, risk_level VARCHAR(16) NOT NULL, risk_reason VARCHAR(255) NOT NULL) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS permission_snapshot (id BIGINT PRIMARY KEY, account_name VARCHAR(64) NOT NULL, object_name VARCHAR(128) NOT NULL, privilege_name VARCHAR(64) NOT NULL, expected_allowed TINYINT(1) NOT NULL, risk_level VARCHAR(16) NOT NULL, CONSTRAINT fk_permission_account FOREIGN KEY (account_name) REFERENCES account_inventory(account_name)) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS web_access_log (id BIGINT PRIMARY KEY, event_time DATETIME NOT NULL, source_ip VARCHAR(45) NOT NULL, path VARCHAR(255) NOT NULL, request_excerpt VARCHAR(500) NOT NULL, response_status INT NOT NULL, risk_label VARCHAR(64) NOT NULL) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS db_audit_log (id BIGINT PRIMARY KEY, event_time DATETIME NOT NULL, account_name VARCHAR(64) NOT NULL, source_ip VARCHAR(45) NOT NULL, statement_summary VARCHAR(500) NOT NULL, affected_object VARCHAR(128) NOT NULL, row_estimate INT NOT NULL, risk_label VARCHAR(64) NOT NULL) ENGINE=InnoDB;
```

```powershell
param(

$EnvFile = Join-Path $ProjectRoot '.env'
$ProjectName = 'shuqi-db-agent'
$MysqlContainer = 'shuqi-mysql-sandbox'
$MysqlVolume = 'shuqi-db-agent-mysql-data'
$ProtectedMySqlPath = 'E:\MySql'

function Assert-ProjectRoot {
  foreach ($item in @('04_开发日志.md','05_开发环境与教学沙箱设计.md','infra\compose.yaml')) {
    if (-not (Test-Path -LiteralPath (Join-Path $ProjectRoot $item))) { throw "Unsafe project root: missing $item" }
  }
  if ((Get-Content -LiteralPath $ComposeFile -Raw -Encoding UTF8).Contains($ProtectedMySqlPath)) {
    throw 'Compose references protected MySQL57 path'
  }
}
function Assert-DockerReady { docker info *> $null; if ($LASTEXITCODE -ne 0) { throw 'Docker Desktop is not ready' } }
function Ensure-EnvFile {
  if (Test-Path -LiteralPath $EnvFile) { return }
  $root = [Convert]::ToHexString([Security.Cryptography.RandomNumberGenerator]::GetBytes(24)).ToLowerInvariant()
  $app = [Convert]::ToHexString([Security.Cryptography.RandomNumberGenerator]::GetBytes(24)).ToLowerInvariant()
  $reader = [Convert]::ToHexString([Security.Cryptography.RandomNumberGenerator]::GetBytes(24)).ToLowerInvariant()
  $lines = @('SANDBOX_MYSQL_PORT=3307',"SANDBOX_MYSQL_ROOT_PASSWORD=$root","SANDBOX_MYSQL_APP_PASSWORD=$app","SANDBOX_MYSQL_READER_PASSWORD=$reader")
  [IO.File]::WriteAllText($EnvFile, ($lines -join [Environment]::NewLine) + [Environment]::NewLine, [Text.UTF8Encoding]::new($false))
}
function Invoke-Compose([string[]]$ComposeArgs) {
  & docker compose --project-name $ProjectName --env-file $EnvFile -f $ComposeFile @ComposeArgs
  if ($LASTEXITCODE -ne 0) { throw "Compose failed: $($ComposeArgs -join ' ')" }
}
function Assert-Port3307AvailableOrOwned {
  $listen = Get-NetTCPConnection -LocalPort 3307 -State Listen -ErrorAction SilentlyContinue
  if (-not $listen) { return }
  $owner = docker ps --filter "name=^/$MysqlContainer$" --format '{{.Names}}'
  if ($owner -ne $MysqlContainer) { throw 'Port 3307 is occupied by a non-project process' }
}
function Wait-MysqlHealthy {
  $deadline = (Get-Date).AddMinutes(3)
  do {
    if ((docker inspect --format '{{.State.Health.Status}}' $MysqlContainer 2>$null) -eq 'healthy') { return }
    Start-Sleep -Seconds 3
  } while ((Get-Date) -lt $deadline)
  throw 'MySQL sandbox health timeout'
}
function Assert-DestructiveTargets {
  Assert-ProjectRoot
  $container = docker ps -a --filter "name=^/$MysqlContainer$" --format '{{.Names}}'
  if ($container -and $container -ne $MysqlContainer) { throw 'Unexpected container target' }
  $volume = docker volume ls --filter "name=^$MysqlVolume$" --format '{{.Name}}'
  if ($volume -and $volume -ne $MysqlVolume) { throw 'Unexpected volume target' }
}

Set-Location -LiteralPath $ProjectRoot
Assert-ProjectRoot
switch ($Action) {

  'Stop' { Assert-DockerReady; Ensure-EnvFile; Invoke-Compose @('down') }
```

### 环境阶段完整回滚

DEV-035/Task 3 的回滚仍保留为当时只退 Task 2/Task 1 的历史流程。如今后用户明确要求撤销整个环境阶段，从 `F:\project_shuqi` 使用下面这一套 fail-closed、可复制的普通反向提交流程：

```powershell
$ErrorActionPreference = 'Stop'
$base = '25351d020a9ef413d9288010028acba579fe7938'
$marker = '^Resolve final Dev Container review findings$'
$logPath = '01_db-security-ops-teaching-agent/04_开发日志.md'

if (@(git status --porcelain).Count -ne 0) { throw 'Rollback requires a clean worktree' }
$preRollbackHead = git rev-parse HEAD
if ($LASTEXITCODE -ne 0 -or [string]::IsNullOrWhiteSpace($preRollbackHead)) { throw 'Cannot resolve pre-rollback HEAD' }
$markerMatches = @(git log --format='%H%x09%s' --grep=$marker)
if ($LASTEXITCODE -ne 0 -or $markerMatches.Count -ne 1) { throw 'Rollback marker must be unique' }
$rollbackHead = ($markerMatches[0] -split "`t", 2)[0]
git merge-base --is-ancestor $base $rollbackHead
if ($LASTEXITCODE -ne 0) { throw 'Baseline is not an ancestor of rollback head' }
git merge-base --is-ancestor $rollbackHead $preRollbackHead
if ($LASTEXITCODE -ne 0) { throw 'Rollback head is not on current history' }
$merges = @(git rev-list --merges "$base..$preRollbackHead")
if ($LASTEXITCODE -ne 0 -or $merges.Count -ne 0) { throw 'Rollback range is not linear' }

$mysqlBefore = docker inspect shuqi-mysql-sandbox | ConvertFrom-Json
if ($LASTEXITCODE -ne 0 -or $mysqlBefore.State.Health.Status -ne 'healthy') { throw 'MySQL baseline failed' }
$mysqlIdBefore = $mysqlBefore.Id
$volumeBefore = @($mysqlBefore.Mounts | Where-Object Destination -eq '/var/lib/mysql')
if ($volumeBefore.Count -ne 1 -or $volumeBefore[0].Name -ne 'shuqi-db-agent-mysql-data') { throw 'Teaching volume baseline failed' }
docker volume inspect shuqi-db-agent-mysql-data | Out-Null
if ($LASTEXITCODE -ne 0) { throw 'Teaching volume missing' }
$mysql57Before = Get-Service MySQL57
$mysql57PidBefore = (Get-CimInstance Win32_Service -Filter "Name='MySQL57'").ProcessId
if ($mysql57Before.Status -ne 'Running') { throw 'MySQL57 baseline failed' }
$portsBefore = @((Get-NetTCPConnection -State Listen -LocalPort 3306,3307) | ForEach-Object { "$($_.LocalAddress)|$($_.LocalPort)|$($_.OwningProcess)" } | Sort-Object)
if (-not ($portsBefore -match '^127\.0\.0\.1\|3307\|') -or -not ($portsBefore -match '\|3306\|')) { throw 'Port baseline failed' }

$commits = @(git rev-list "$base..$rollbackHead")
if ($LASTEXITCODE -ne 0 -or $commits.Count -eq 0) { throw 'Rollback list failed' }
git revert --no-commit $commits
if ($LASTEXITCODE -ne 0) {
    git revert --abort
    throw 'Revert failed; abort completed, stop here'
}
git restore --source=$preRollbackHead --staged --worktree -- 01_db-security-ops-teaching-agent/04_开发日志.md
if ($LASTEXITCODE -ne 0) { throw 'Development log restore failed' }
$confirmation = Read-Host "Use an editor to append the actual rollback record to $logPath, then type ROLLBACK-LOG-APPENDED"
if ($confirmation -cne 'ROLLBACK-LOG-APPENDED') { throw 'Rollback log was not confirmed' }
git add -- $logPath
if ($LASTEXITCODE -ne 0) { throw 'Cannot stage rollback log' }
git diff --cached --quiet -- $logPath
if ($LASTEXITCODE -eq 0) { throw 'No appended rollback record detected' }
if ($LASTEXITCODE -ne 1) { throw 'Cannot validate rollback log' }

docker compose --project-name shuqi-db-agent --env-file 01_db-security-ops-teaching-agent/.env -f 01_db-security-ops-teaching-agent/infra/compose.yaml config --quiet
if ($LASTEXITCODE -ne 0) { throw 'Compose config failed' }
docker compose --project-name shuqi-db-agent --env-file 01_db-security-ops-teaching-agent/.env -f 01_db-security-ops-teaching-agent/infra/compose.yaml up -d --build --no-deps --force-recreate workspace
if ($LASTEXITCODE -ne 0) { throw 'Workspace-only rebuild failed' }
docker exec -u vscode shuqi-workspace pytest -q
if ($LASTEXITCODE -ne 0) { throw 'Post-rollback pytest failed' }

$mysqlAfter = docker inspect shuqi-mysql-sandbox | ConvertFrom-Json
$volumeAfter = @($mysqlAfter.Mounts | Where-Object Destination -eq '/var/lib/mysql')
if ($LASTEXITCODE -ne 0 -or $mysqlAfter.Id -ne $mysqlIdBefore -or $mysqlAfter.State.Health.Status -ne 'healthy') { throw 'MySQL ID/health changed' }
if ($volumeAfter.Count -ne 1 -or $volumeAfter[0].Name -ne $volumeBefore[0].Name) { throw 'Teaching volume changed' }
docker volume inspect shuqi-db-agent-mysql-data | Out-Null
if ($LASTEXITCODE -ne 0) { throw 'Teaching volume missing after rollback' }
$mysql57After = Get-Service MySQL57
$mysql57PidAfter = (Get-CimInstance Win32_Service -Filter "Name='MySQL57'").ProcessId
if ($mysql57After.Status -ne 'Running' -or $mysql57PidAfter -ne $mysql57PidBefore) { throw 'MySQL57 changed' }
$portsAfter = @((Get-NetTCPConnection -State Listen -LocalPort 3306,3307) | ForEach-Object { "$($_.LocalAddress)|$($_.LocalPort)|$($_.OwningProcess)" } | Sort-Object)
if (Compare-Object $portsBefore $portsAfter) { throw 'Port listeners changed' }

git commit -m 'Rollback complete environment stage'
if ($LASTEXITCODE -ne 0) { throw 'Rollback commit failed' }
git push origin main
if ($LASTEXITCODE -ne 0) { throw 'Rollback push failed' }
$local = git rev-parse HEAD
$remoteLines = @(git ls-remote origin refs/heads/main)
if ($LASTEXITCODE -ne 0 -or $remoteLines.Count -ne 1) { throw 'Remote SHA lookup failed' }
$remote = ($remoteLines[0] -split '\s+')[0]
if ([string]::IsNullOrWhiteSpace($local) -or $local -ne $remote) { throw 'Local/remote SHA mismatch' }
```
